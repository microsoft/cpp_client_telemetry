cmake_minimum_required(VERSION 3.19)
project(sqlite_coexistence_test LANGUAGES C CXX)

get_filename_component(SDK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MATSDK_SQLITE_PROVIDER MINIMAL CACHE STRING "")
set(MATSDK_ZLIB_PROVIDER VENDORED CACHE STRING "")
add_library(external_sqlite STATIC "${SDK_ROOT}/sqlite/sqlite3.c")
target_include_directories(external_sqlite PUBLIC "${SDK_ROOT}/sqlite")
if(NOT WIN32)
  find_package(Threads REQUIRED)
  target_link_libraries(external_sqlite PUBLIC Threads::Threads "${CMAKE_DL_LIBS}" m)
endif()

if(MATSDK_INSTALLED_PREFIX)
  find_package(MSTelemetry CONFIG REQUIRED PATHS "${MATSDK_INSTALLED_PREFIX}"
    NO_DEFAULT_PATH)
else()
  foreach(option HEADERS LIBRARY)
    set(MATSDK_BUILD_${option} ON CACHE BOOL "" FORCE)
  endforeach()
  foreach(option UNIT_TESTS FUNC_TESTS TEST_TOOL PACKAGE JNI_WRAPPER OBJC_WRAPPER SWIFT_WRAPPER)
    set(MATSDK_BUILD_${option} OFF CACHE BOOL "" FORCE)
  endforeach()
  add_subdirectory("${SDK_ROOT}" sdk)
endif()

add_executable(sqlite_coexistence main.cpp)
target_link_libraries(sqlite_coexistence PRIVATE MSTelemetry::mat external_sqlite)
get_target_property(library_type MSTelemetry::mat TYPE)
if(library_type STREQUAL "STATIC_LIBRARY")
  target_compile_definitions(sqlite_coexistence PRIVATE MATSDK_TEST_PRIVATE_SQLITE)
endif()
if(MATSDK_SQLITE_PROVIDER STREQUAL "MINIMAL")
  target_compile_definitions(sqlite_coexistence PRIVATE MATSDK_TEST_MINIMAL_SQLITE)
endif()
if(WIN32 AND library_type STREQUAL "SHARED_LIBRARY")
  add_custom_command(TARGET sqlite_coexistence POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different
      "$<TARGET_FILE:MSTelemetry::mat>" "$<TARGET_FILE_DIR:sqlite_coexistence>")
endif()
if(MSVC)
  target_compile_options(sqlite_coexistence PRIVATE /W4 /WX)
else()
  target_compile_options(sqlite_coexistence PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME sqlite_coexistence COMMAND sqlite_coexistence)
