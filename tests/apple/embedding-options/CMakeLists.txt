cmake_minimum_required(VERSION 3.19)
project(apple_embedding_options LANGUAGES C CXX)

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/platform.cmake" "set(APPLE ${TEST_APPLE})\n")
set(CMAKE_PROJECT_INCLUDE "${CMAKE_CURRENT_BINARY_DIR}/platform.cmake")
set(APPLE "${TEST_APPLE}")
if(TEST_VCPKG)
  set(VCPKG_TARGET_TRIPLET test-triplet)
endif()
set(MATSDK_BUILD_HEADERS OFF CACHE BOOL "" FORCE)
set(MATSDK_BUILD_LIBRARY OFF CACHE BOOL "" FORCE)
set(MATSDK_BUILD_TEST_TOOL OFF CACHE BOOL "" FORCE)
set(MATSDK_BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(MATSDK_BUILD_FUNC_TESTS OFF CACHE BOOL "" FORCE)
set(MATSDK_BUILD_PACKAGE OFF CACHE BOOL "" FORCE)
set(MATSDK_BUILD_APPLE_HTTP "${TEST_HTTP}" CACHE BOOL "" FORCE)
set(MATSDK_SQLITE_PROVIDER SYSTEM CACHE STRING "" FORCE)
set(MATSDK_ZLIB_PROVIDER SYSTEM CACHE STRING "" FORCE)
add_library(CURL::libcurl INTERFACE IMPORTED)
add_library(nlohmann_json::nlohmann_json INTERFACE IMPORTED)
if(TEST_PROVIDED OR NOT TEST_APPLE)
  add_library(SQLite3::SQLite3 INTERFACE IMPORTED)
  add_library(ZLIB::ZLIB INTERFACE IMPORTED)
  set_property(TARGET SQLite3::SQLite3 PROPERTY INTERFACE_LINK_LIBRARIES caller-sqlite)
  set_property(TARGET ZLIB::ZLIB PROPERTY INTERFACE_LINK_LIBRARIES caller-zlib)
endif()
foreach(package SQLite3 unofficial-sqlite3 ZLIB nlohmann_json)
  set(CMAKE_DISABLE_FIND_PACKAGE_${package} ON)
endforeach()
add_subdirectory("${SDK_ROOT}" sdk)
foreach(pair "SQLite3::SQLite3|sqlite3|caller-sqlite" "ZLIB::ZLIB|z|caller-zlib")
  string(REPLACE "|" ";" pair "${pair}")
  list(GET pair 0 target)
  list(GET pair 1 expected)
  if(TEST_PROVIDED OR NOT TEST_APPLE)
    list(GET pair 2 expected)
  endif()
  get_target_property(links "${target}" INTERFACE_LINK_LIBRARIES)
  if(NOT links STREQUAL expected)
    message(FATAL_ERROR "${target} links ${links}, expected ${expected}")
  endif()
endforeach()

# Evaluate the production target guard without requiring an Apple compiler or SDK.
file(READ "${SDK_ROOT}/lib/CMakeLists.txt" library)
string(FIND "${library}" "if(APPLE AND MATSDK_BUILD_APPLE_HTTP)" start)
if(start EQUAL -1)
  message(FATAL_ERROR "Update Apple delegate target instrumentation")
endif()
string(SUBSTRING "${library}" ${start} -1 definition)
string(FIND "${definition}" "\nendif()" end)
math(EXPR end "${end} + 8")
string(SUBSTRING "${definition}" 0 ${end} definition)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/delegate.cpp" "")
add_library(mat STATIC "${CMAKE_CURRENT_BINARY_DIR}/delegate.cpp")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/delegate.cmake" "${definition}")
include("${CMAKE_CURRENT_BINARY_DIR}/delegate.cmake")
get_target_property(private mat COMPILE_DEFINITIONS)
get_target_property(public mat INTERFACE_COMPILE_DEFINITIONS)
set(expected "MATStreamingSessionDelegate=${MATSDK_APPLE_HTTP_DELEGATE_CLASS}")
if(TEST_APPLE AND TEST_HTTP)
  if(NOT expected IN_LIST private)
    message(FATAL_ERROR "Missing private Apple delegate definition: ${private}")
  endif()
elseif(expected IN_LIST private)
  message(FATAL_ERROR "Delegate definition leaked into a non-Apple HTTP build")
endif()
if(expected IN_LIST public)
  message(FATAL_ERROR "Delegate definition propagated to consumers")
endif()
