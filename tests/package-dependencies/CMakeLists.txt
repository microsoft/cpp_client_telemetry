cmake_minimum_required(VERSION 3.19)
project(package_dependency_test LANGUAGES NONE)

set(CMAKE_SYSTEM_NAME Darwin)
set(CMAKE_FIND_PACKAGE_PREFER_CONFIG ON)
set(CMAKE_PREFIX_PATH "${CMAKE_CURRENT_BINARY_DIR}/packages")
file(MAKE_DIRECTORY "${CMAKE_PREFIX_PATH}")
foreach(package_target IN ITEMS
    "SQLite3|SQLite::SQLite3" "unofficial-sqlite3|unofficial::sqlite3::sqlite3"
    "ZLIB|ZLIB::ZLIB" "CURL|CURL::libcurl")
  string(REPLACE "|" ";" pair "${package_target}")
  list(GET pair 0 package)
  list(GET pair 1 target)
  file(WRITE "${CMAKE_PREFIX_PATH}/${package}Config.cmake"
    "add_library(${target} INTERFACE IMPORTED GLOBAL)\n"
    "set_property(TARGET ${target} PROPERTY INTERFACE_LINK_LIBRARIES package-${package})\n")
endforeach()
file(WRITE "${CMAKE_PREFIX_PATH}/OpenSSLConfig.cmake"
  "add_library(OpenSSL::SSL INTERFACE IMPORTED GLOBAL)\n"
  "add_library(OpenSSL::Crypto INTERFACE IMPORTED GLOBAL)\n")

set(MATSDK_CONFIG_STATIC_PACKAGE ON)
if(SHARED)
  set(MATSDK_CONFIG_STATIC_PACKAGE OFF)
endif()
set(MATSDK_SQLITE_DEPENDENCY_KIND "${SQLITE_KIND}")
set(MATSDK_ZLIB_DEPENDENCY_KIND "${ZLIB_KIND}")
set(MATSDK_SQLITE_PACKAGE_NAME SQLite3)
set(MATSDK_SQLITE_PACKAGE_TARGET SQLite::SQLite3)
if(UNOFFICIAL_SQLITE)
  set(MATSDK_SQLITE_PACKAGE_NAME unofficial-sqlite3)
  set(MATSDK_SQLITE_PACKAGE_TARGET unofficial::sqlite3::sqlite3)
endif()
if(NOT DEFINED CURL_KIND)
  set(CURL_KIND PACKAGE)
endif()
if(NOT SHARED)
  foreach(pair IN ITEMS "SQLite3::SQLite3|${SQLITE_KIND}" "ZLIB::ZLIB|${ZLIB_KIND}"
      "CURL::libcurl|${CURL_KIND}")
    string(REPLACE "|" ";" pair "${pair}")
    list(GET pair 0 target)
    list(GET pair 1 kind)
    if(kind STREQUAL "PROVIDED" AND NOT target STREQUAL MISSING_TARGET)
      if(target STREQUAL "SQLite3::SQLite3" AND LEGACY_SQLITE)
        set(target SQLite::SQLite3)
      endif()
      add_library("${target}" INTERFACE IMPORTED GLOBAL)
      set_property(TARGET "${target}" PROPERTY INTERFACE_LINK_LIBRARIES provided)
    endif()
  endforeach()
endif()
set(MATSDK_NEEDS_CURL ON)
set(MATSDK_CURL_DEPENDENCY_KIND "${CURL_KIND}")
set(MATSDK_CURL_TLS_BACKEND_UPPER MBEDTLS)
if(OPENSSL)
  set(MATSDK_CURL_TLS_BACKEND_UPPER OPENSSL)
endif()
set(CMAKE_INSTALL_LIBDIR lib)
set(CMAKE_INSTALL_DATADIR share)
set(MATSDK_ANDROID_HTTP_CLIENT_RESOLVED NONE)
set(PACKAGE_INIT
  "set(PACKAGE_PREFIX_DIR \"${CMAKE_CURRENT_BINARY_DIR}\")\nmacro(check_required_components name)\nendmacro()")
file(COPY "${SDK_ROOT}/cmake/MatsdkDependencyTargets.cmake"
  "${SDK_ROOT}/cmake/MatsdkAppleSystemDeps.cmake"
  DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/MSTelemetryTargets.cmake"
  "add_library(MSTelemetry::mat INTERFACE IMPORTED)\n")
configure_file("${SDK_ROOT}/cmake/MSTelemetryConfig.cmake.in"
  "${CMAKE_CURRENT_BINARY_DIR}/MSTelemetryConfig.cmake" @ONLY)
if(MISSING_PACKAGE)
  set(CMAKE_DISABLE_FIND_PACKAGE_${MISSING_PACKAGE} ON)
endif()
set(lookup_options CONFIG)
if(LOOKUP_MODE STREQUAL "REQUIRED")
  list(APPEND lookup_options REQUIRED)
elseif(LOOKUP_MODE STREQUAL "QUIET")
  list(APPEND lookup_options QUIET)
endif()
find_package(MSTelemetry ${lookup_options}
  PATHS "${CMAKE_CURRENT_BINARY_DIR}" NO_DEFAULT_PATH)
if(MISSING_PACKAGE AND NOT SHARED)
  if(MSTelemetry_FOUND OR TARGET MSTelemetry::mat)
    message(FATAL_ERROR "SDK package was imported despite missing ${MISSING_PACKAGE}")
  endif()
  if(NOT MSTelemetry_NOT_FOUND_MESSAGE MATCHES "${MISSING_PACKAGE}")
    message(FATAL_ERROR "Missing dependency diagnostic lost: ${MSTelemetry_NOT_FOUND_MESSAGE}")
  endif()
  return()
endif()
if(NOT MSTelemetry_FOUND OR NOT TARGET MSTelemetry::mat)
  message(FATAL_ERROR "SDK package was not imported")
endif()

foreach(pair IN ITEMS
    "SQLite3::SQLite3|${SQLITE_KIND}|MSTelemetry::sqlite_dependency"
    "ZLIB::ZLIB|${ZLIB_KIND}|MSTelemetry::zlib_dependency"
    "CURL::libcurl|${CURL_KIND}|MSTelemetry::curl_dependency")
  string(REPLACE "|" ";" pair "${pair}")
  list(GET pair 0 target)
  list(GET pair 1 kind)
  list(GET pair 2 wrapper)
  if(NOT SHARED AND kind STREQUAL "BUNDLED" AND target STREQUAL "CURL::libcurl")
    get_target_property(links "${wrapper}" INTERFACE_LINK_LIBRARIES)
    if(NOT links STREQUAL "MSTelemetry::curl_archive")
      message(FATAL_ERROR "Bundled curl was replaced by ${links}")
    endif()
    get_target_property(tls_links MSTelemetry::curl_archive INTERFACE_LINK_LIBRARIES)
    if(OPENSSL)
      set(expected_tls "OpenSSL::SSL;OpenSSL::Crypto")
    else()
      set(expected_tls "MSTelemetry::mbedtls;MSTelemetry::mbedx509;MSTelemetry::mbedcrypto")
    endif()
    if(NOT tls_links STREQUAL expected_tls)
      message(FATAL_ERROR "Bundled curl TLS was replaced by ${tls_links}")
    endif()
  elseif(SHARED OR kind MATCHES "^(BUNDLED|NONE)$")
    if(TARGET "${wrapper}")
      message(FATAL_ERROR "Unexpected private dependency ${wrapper}")
    endif()
  else()
    if(NOT TARGET "${wrapper}" OR NOT TARGET "${target}")
      message(FATAL_ERROR "Missing ${kind} dependency ${target}")
    endif()
    get_target_property(links "${target}" INTERFACE_LINK_LIBRARIES)
    if(kind STREQUAL "APPLE_SYSTEM")
      if(NOT links MATCHES "^(sqlite3|z)$")
        message(FATAL_ERROR "Apple dependency was replaced by ${links}")
      endif()
    elseif(kind STREQUAL "PROVIDED")
      if(NOT links MATCHES "^(provided|SQLite::SQLite3)$")
        message(FATAL_ERROR "Caller dependency was replaced by ${links}")
      endif()
    elseif(NOT links MATCHES "^(package-|SQLite::SQLite3|unofficial::sqlite3::sqlite3)")
      message(FATAL_ERROR "Package dependency was replaced by ${links}")
    endif()
  endif()
endforeach()
